# 유니온 타입
- 바(`|`) 기호를 기준으로 기재된 타입 중 하나를 만족하면 허용하는 타입 구조 (교집합)
- 다중 타입 허용 ─ 숫자나 문자열 등 여러 타입의 데이터가 입력될 수 있는 변수를 정의할 때 사용
- 접근 제한 원리 ─ 서로 다른 객체 타입들을 묶으면 컴파일러는 모든 타입에 공통으로 존재하는 속성만 접근 허용
	- 특정 객체 타입에만 존재하는 속성에 접근할 때 발생할 수 있는 런타임 오류를 방지해 안전성 보장
```ts
// 카드 결제 정보를 나타내는 인터페이스 정의
interface CardPayment {
    amount: number; // 결제 금액 (공통 속성)
    cardNumber: string; // 카드 번호 (전용 속성)
}

// 계좌이체 결제 정보를 나타내는 인터페이스 정의
interface BankTransfer {
    amount: number; // 결제 금액 (공통 속성)
    bankCode: string; // 은행 코드 (전용 속성)
}

// 유니온 타입 정의
type PaymentMethod = CardPayment | BankTransfer;

// in 연산자를 사용한 타입 내로잉
function processPayment(payment: PaymentMethod): void {
    // 여기: payment 는 CardPayment | BankTransfer (공통인 amount만 접근 가능)
    console.log("결제요청금액:", payment.amount);
    // console.log("카드번호:", payment.cardNumber); → 오류코드
    
    if ("cardNumber" in payment) {              // 타입 내로잉으로는 접근 가능 → 오류 없음
	    // 여기: payment 는 CardPayment 로 좁혀짐
	    console.log("카드번호:", payment.cardNumber);
    } else {
	    // 여기: payment 는 BankTransfer 로 좁혀짐
	    console.log("은행코드:", payment.bankCode);
    }
}

// 카드 결제 객체 정의
const myPayment: CardPayment = {
    amount: 50000,
    cardNumber: "1111-1111-1111-1111",
};

// 유니온 타입을 매개변수로 받는 processPayment 함수 호출
processPayment(myPayment);
// myPayment가 실제로 cardPayment여도 상관 없다. 
// processPayment 함수를 선언할 때 cardPayment, bankTransfer를 모두 받을 수 있게 정의했기 때문
```
# 인터섹션 타입
- 앰퍼샌드(`&`) 기호를 사용하여 지정된 모든 타입의 속성을 동시에 만족하도록 합성하는 타입
	- 기존에 정의된 독립적인 타입들을 결합하여 확장된 타입을 생성할 때 활용한다. 
```ts
// 객체에 쓸 때: 속성을 합침
type A = { name: string };
type B = { age: number };
type C = A & B;   // { name: string; age: number }

// 원시에 쓸 때: never (동시에 여러 원시 타입일 수 없으므로) → 에러는 아니지만 무쓸모
type X = string & number;

// 유니온과 섞어 쓸 때 
type T = ("a" | "b" | "c") & ("b" | "c" | "d"); // "b" | "c"

// 원시 타입에 표식을 붙이는 브랜드 타입으로 활용
// string과 구분되는 타입을 만드는 패턴
type UserId = string & { readonly __brand: "UserId" };
```

|              | `extends`                          | `&`                   |
| ------------ | ---------------------------------- | --------------------- |
| 대상           | `interface`끼리 (type 별칭도 객체 형태면 가능) | 어떤 타입이든               |
| 같은 속성이 충돌할 때 | 컴파일 에러                             | 에러 없이 `never`가 될 수 있음 |
| 유니온과 함께      | 불가                                 | 가능                    |

> [!NOTE] 인터페이스의 객체 타입 합성과 확장
> - 인터페이스 객체 구조 확장은 주로 `extends`를 사용하지만 `&`도 사용할 수 있다. 
> 	- *cf.* `extends`는 단순히 합치는 개념이 아니라 할당 가능 여부를 묻는 개념이다.
> - 그럼에도 `extends`를 사용하는 이유는
> 	- 같은 속성의 타입이 다르면 `never`로 만드는 `&`와 달리 `extends`는 에러를 발생시킨다. 
> 	- 상속 관계를 코드로 바로 읽을 수 있어서 의도를 명확히 할 수 있다. 
> 	- 에러 메세지가 이름이 있는 인터페이스로 표시되어 디버깅에 용이하다. 
> 	- 컴파일러가 `extend`된 인터페이스는 캐싱해서 처리하기 때문에 `&`보다 성능에 유리하다. 
> - (`&`를 사용해야만 하는 경우 (`type`으로 정의한 타입 간 결합) 외에) 부득이 `&`를 사용해야 하는 경우는
> 	- 유니온을 섞어서 사용해야 할 때
> 	- 원시 타입에 붙여 브랜드 타입을 정의할 때
> 	- 이름 있는 새 타입을 만들 필요 없이 일회성으로 합성할 때

# 브랜드 타입
- 구조가 같아서 타입스크립트가 구분하지 못하는 타입에 꼬리표(brand)를 붙여서 구분되게 만드는 기법
	- 타입스크립트는 덕타이핑(구조적 타이핑)이기 때문에 구조가 같으면 같은 타입으로 취급하기 때문에 필요
- 사용 예시 ─ 똑같은 Id 구조 → 같은 타입으로 취급해서 서로 바뀌어도 에러 없이 통과될 가능성이 있을 때
```ts
// 일반(원시) 타입 정의
type UserId = string;
type OrderId = string;

function getUser(id: UserId) {  }

const orderId: OrderId = "order-123";
getUser(orderId);   // 에러 없음! 둘 다 그냥 string 이라서

// 브랜드 타입 정의
type UserId  = string & { readonly __brand: "UserId" };
type OrderId = string & { readonly __brand: "OrderId" };

function getUser(id: UserId) {  }

declare const userId: UserId;
declare const orderId: OrderId;

getUser(userId);    // OK
getUser(orderId);   // 에러: OrderId 는 UserId 에 할당 불가
getUser("abc");     // 에러: 그냥 string 도 불가

// 브랜드 타입 정의 후 값을 만드는 방법
function toUserId(raw: string): UserId {
    // 필요하면 여기서 형식 검증도 가능
    return raw as UserId;
}

const id = toUserId("u-001");   // UserId
getUser(id);                    // OK
```
# 리터럴 타입

- string이나 number 타입의 전체 범위 대신 지정된 값만 허용하는 타입 정의 방식 (특정 값으로 범위 제한)
- `let`이나 매개변수, 속성 등 바뀔 수 있는 자리의 허용 값을 좁히기 위해 리터럴 타입(유니온 타입 결합)을 사용
	- `const`는 값이 변하지 않기 때문에 리터럴 타입으로 제한할 의미가 적다. 
		- *cf.* 반면에 값이 실행 중에 결정되거나 함수 반환값, 알 수 없는 외부 입력, 배열 등에는 유니온을 자주 사용
```ts
// let은 넓은 타입(string)으로 자동 추론한다. → 리터럴 타입은 별도로 직접 지정해야 한다. 
// 나중에 다른 값으로 바뀔 수 있으니까 타입 추론을 넓게
let a = "hi";   

// const는 리터럴 타입("hi")으로 자동 추론한다. 
// 값이 바뀌지 않으니까 정확한 값 그대로 → 이 경우, 이 자체로 리터럴 타입 ("hi"로 허용 값 지정)
const b = "hi"; 

// let을 활용한 특정 값 범위 제한 예시
let a: "on" = "on";              // 리터럴 타입을 직접 지정
let b: "on" | "off" = "on";      // 유니온으로 허용 값을 늘림
b = "off";                        // 에러 없음
b = "idle";                       // 에러
```
# Keyof 연산자
- 객체 형태의 타입 앞에 작성 → 해당 객체의 속성명들을 유니온 타입으로 추출
- 오타 등으로 인해 존재하지 않는 속성에 접근하는 실수를 방지한다. 
```ts
// 사용자 프로필 정보를 나타내는 인터페이스 선언
interface UserProfile {
    id: number;
    name: string;
    email: string;
    age: number;
}

// keyof 연산자로 UserProfile의 속성 키만 추출
type UserKeys = keyof UserProfile;
const a: UserKeys = "name";
const b: UserKeys = "Name"; // 에러: 대소문자를 구분하므로 존재하지 않는 속성에 해당
```
# 판별자
- 모든 멤버가 공통으로 가진 속성으로, 리터럴 타입이며 멤버별로 고유한 객체 타입을 구분하는 속성을 가진다. 
	- *cf.* 멤버 ─ 클래스의 필드, 객체 또는 타입, 인터페이스의 속성, 메서드 등을 합쳐 부르는 말
- 판별에 필요한 정보를 타입 설계 단계에서 미리 심어두고 필요한 시점에 값 하나로 멤버를 확정할 수 있다. 
	- 판별을 위한 별도의 검사 코드(타입 가드) 필요 없이 정의해 둔 판별자만으로 타입 내로잉 가능 
## 서로소 판별 유니온
- 필요 조건 ─ 아래 조건 중 하나라도 위반하는 경우 판별(타입 내로잉) 불가
	1. 모든 멤버에 공통 속성이 있어야 한다.
	2. 그 속성의 타입이 리터럴이어야 한다. 
	3. 각 속성의 값이 멤버마다 겹치치 않아야 한다. (서로소)
- `in` 연산자를 활용한 타입 내로잉과의 차이점
	- `in` 연산자 타입 가드 ─ 키 보유 여부로 타입 내로잉
	- 서로소 판별 유니온 타입 가드 ─ 공통 속성의 값 비교로 타입 내로잉 → 중복되지 않으므로 정확히 구분 가능
```ts
/* type을 공통 속성으로 부여합니다. */
interface CardPayment   { type: "card"; amount: number; cardNumber: string }
interface BankTransfer  { type: "bank"; amount: number; bankCode: string }

// 서로소 판별 유니온을 활용한 타입 내로잉
if (payment.type === "card") {
    payment.cardNumber; // OK
}
```

|        | `in` 검사              | 판별 유니온                                   |
| ------ | -------------------- | ---------------------------------------- |
| **기준** | 속성 존재 여부             | 공통 속성의 리터럴 값 (설계해 둔 태그 값)                |
| **장점** | 타입을 따로 고치지 않아도 됨     | 명확하고, 멤버가 늘어도 `switch`와 `never`로 처리하기 쉬움 |
| **주의** | 선택 속성(`?`)이 섞이면 모호해짐 | 판별자를 직접 타입에 넣어야 함                        |
# 제네릭(Generic)
- 동적 자료형 명시 ─ 다양한 타입을 하나의 틀로 다루면서 타입 안전성을 보장하는 타입 변수 시스템
	- 주로 함수 호출 또는 인터페이스 에서 사용하고 제약이 없는 경우 타입 내로잉이 필요하다. 
- `unknown`과 마찬가지로 `any` 타입의 한계를 해소할 수 있다. 
	- 타입을 미리 고정하지 않고 함수나 인터페이스를 호출하는 시점에 타입을 인자로 받아 결정한다. 
	- **제약 없는 제네릭**은 `unknown`과 같다. 
		- 제네릭 함수가 뭔지 모르기 때문에 내로잉 필요 → 함수 안에서 직접 검사
	- **제약 있는 제네릭**은 최소한의 타입을 보장한다. 
		- 내로잉 불필요 (타입 가드 없이 바로 쓸 수 있음) → 함수 안에서 검사하지 않고 호출할 때 검사
		- `<T extends 제한타입>` 형태 ─ `T`가 반드시 해당 타입을 만족하도록 강제 (`T`는 제한타입의 하위 타입)
		- *e.g.* `<T extends LengthWise>` ─ `T`는 `lengthWise` 조건을 만족하는 타입만 허용된다. 
		- `<T, K extends keyof T>` 형태 ─ K 타입이 T 객체의 키 중 하나를 만족하도록 강제
	- `unknown`은 입력 타입 정보가 사라지지만 제네릭은 반환 시까지 입력 타입 정보를 유지한다. 
			- 따라서 입력 타입이 출력이나 다른 매개변수 타입과 연결되어야 한다면 제네릭 사용 권장
```ts
/* 함수 안에서 T가 뭔지 모르기 때문에 에러 발생 */
function echo<T>(value: T): T {
  return value.toUpperCase();   // 에러: Property 'toUpperCase' does not exist on type 'T'
}

/* 함수 안에서 T가 뭔지 알도록 지정 */
function echo<T>(value: T): T {
  if (typeof value === "string") {
    value.toUpperCase();   // OK: 이 블록에서 value 는 T & string
  }
  return value;
}

/* 함수 밖에서 extends로 타입 제한 ver.1 */
function echo<T extends string>(value: T): T {
  value.toUpperCase();   // OK: T 는 최소한 string 이라고 보장됨
  return value;
}

/* 합수 밖에서 extends로 타입 제한 ver.2 */
interface Lengthwise { length: number; }
function printLength<T extends Lengthwise>(item: T): number {
    return item.length;
}
console.log(printLength("안녕하세요."));   // OK: string에 length가 있으니까
console.log(printLength([10, 20, 30]));   // OK: 배열에 length가 있으니까
console.log(printLength({ length: 10 })); // OK: length 속성이 있는 객체니까
console.log(printLength(1234));           // 에러: number에는 length가 없으니까

/* 함수 밖에서 extends와 keyof로 타입 제한 */
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];
}
const product = { id: 101, title: "무선 키보드", price: 45000, isAvailable: true };

const title = getProperty(product, "title"); // 존재하는 키
console.log(title);                          // 반환 타입: string

const discount = getProperty(product, "discount"); // 존재하지 않는 키
console.log(discount);                             // 컴파일 에러

/* unknown과의 차이점: 호출한 쪽에서 타입 보존 */
function echoUnknown(value: unknown): unknown { return value; }
function echoGeneric<T>(value: T): T { return value; }

const a = echoUnknown("hi");      // a: unknown (정보를 잃음)
a.toUpperCase()                   // 좁히기 없이 바로 사용 불가 → 타입 에러 발생
if (typeof a === "string") {
  console.log(a.toUpperCase());   // 이 블록에서 a 는 string (좁히기) → "HI"
  console.log(a.length);          // 2
}
const b = echoGeneric("hi");      // b: "hi" (정보가 유지됨)
b.toUpperCase()                   // 좁히기 없이 바로 사용 가능
```
## 제네릭 함수
- 함수를 호출할 때 타입을 인자로 전달해 매개변수와 반환값의 타입을 동적으로 정하는 함수
	- 타입 매개변수 식별자 ─ 관례적으로 <T, U, V>  순으로 사용한다. 
	- 함수 호출 시 처음 타입을 명시하지 않아도 전달되는 인자를 보고 컴파일러가 타입을 추론할 수 있다. 
		- [[2026-10-02#^9e6a20]] ─ 단, `T`가 매개변수 타입에 쓰이는 경우(인자 확인이 가능한 경우)에만 추론 가능
```ts
function echo<T>(value: T): T {
  return value;
}
echo<string>("hi");   // 명시
echo("hi");           // 생략: 컴파일러가 T = string 으로 추론

/* T가 여러 곳에 쓰이면 모든 위치에서 일관되게 맞는 타입을 찾는다. */ 
const a = echo("hi");        // a: "hi"    (const 리터럴 취급)
const b = echo(123);         // b: 123
let   c = echo("hi");        // c: string  (let 이라 넓어짐)
// 근데 일관되게 맞는 타입을 못찾으면 에러가 발생한다. 
function pair<T>(a: T, b: T): T[] { return [a, b]; }
pair(1, 2);                  // T = number
pair(1, "x");                // 에러: T 를 하나로 맞출 수 없음 (일관되게 맞는 타입 추론 불가)

/* 추론이 불가한 경우(매개변수가 아예 없어서 추론 근거가 없는 경우)에는 생략하면 안된다. */
function create<T>(): T[] {
  return [];
}
const x = create();          // T 추론 불가 → unknown[]
const y = create<string>();  // 명시해야 함 → string[]
```

## 서버 응답을 위한 공통 제네릭 인터페이스 정의
- API 데이터는 유동적인 성격이 강하므로 제네릭 인터페이스를 적극 활용할 수 있다. 
```ts
// 공통 제네릭 구조 정의
interface ApiResponse<T> {
    success: boolean;   // 요청 성공 여부 (고정 타입)
    message: string;    // 응답 메시지 (고정 타입)
    data: T;            // 실제 데이터 (동적 타입 매개변수)
}

// 개별 데이터 구조 정의
interface UserProfile {
    id: number;     // 사용자 고유 번호
    name: string;   // 사용자 이름
    email: string;  // 사용자 이메일
}

interface ProductItem {
    code: string;       // 상품 코드
    price: number;      // 상품 가격
}

// UserProfile 타입을 가질 수 있는 응답 객체 생성
const userResponse: ApiResponse<UserProfile> = {
    success: true,
    message: "사용자 정보 조회 성공",
    data: {
        id: 1,
        name: "김철수",
        email: "user01@test.org",
    }
};

// ProductItem 타입을 가질 수 있는 응답 객체 생성
const productResponse: ApiResponse<ProductItem> = {
    success: true,
    message: "상품 정보 조회 성공",
    data: {
        code: "PROD-01101",
        price: 20000,
    }
};
```

> [!NOTE] 덕타이핑 (구조적 타이핑)
> - 상속 또는 `implements` 하지 않아도 정의된 인터페이스 구조를 따르면 해당 인터페이스의 하위 타입으로 인정
> 	- 상속 관계가 아니라 모양(구조)를 보고 판단 (구조만 맞으면 통과)
> 		- 이름은 다르지만 모양이 같은 타입이 서로 호환되므로 의도치 않은 타입이 통과될 수 있어 유의해야 한다.
> 	- *cf.* Java나 C#처럼 `implements`를 꼭 써야 하는 언어는 명목적 타이핑이라고 한다. 

# 클래스 접근 제어자
- 내부 데이터를 외부에 얼마나 공개할지 결정하는 장치 (캡슐화 수준을 결정하는 장치)
	- *cf.* 자바스크립트에서는 접근 제어자가 없어서 `_`, 클로저, 게터/세터로 접근 제어를 흉내내는 식으로 사용
		- ES2022에서 `#`으로 쓰는 진짜 private 필드 문법을 제공해서 타입스크립트의 `private`와 같이 사용 가능
- 인스턴스 변수 , 인스턴스 메서드 앞에 사용해서 접근 범위를 지정하고 컴파일 단계에서 위반 방지
- public ─ 클래스 내외부 및 자식 클래스 어디서든 접근 가능, 기본값이기 때문에 생략 가능
- protected ─ 해당 클래스 내부에서만 접근 가능 → 상속을 통하면 하위 클래스 내부에서도 접근 가능
- private ─ 해당 클래스 내부에서만 접근 가능 → 컴파일 시에 통제하기 때문에 안정적으로 사용 가능
	- 데이터는 자식 인스턴스에도 존재하긴 하지만 컴파일 할 때의 접근만 막을 뿐이고, 객체에 속성은 그대로 있음
		- `private`이더라도 `public`이나 `protected` 메서드로 값을 열어주면 자식에 접근 가능
	- 부모에 `private secret`이 있으면 자식이 `secret`을 선언하면 에러 (자식이 같은 이름으로 재선언 불가)
```ts
class Account {
  public owner = "철수";      // 어디서나 접근 가능 (기본값)
  private balance = 0;        // 클래스 안에서만
  protected id = 1;           // 클래스 + 자식 클래스에서만
}
new Account().balance;        // 에러

/* 접근 제한자가 적용된 계좌 클래스 예제 */
class SecureAccount {
    owner: string;
    protected accountType: string;
    private balance: number;      // private → protected

    constructor(owner: string, accountType: string, balance: number) {
        this.owner = owner;
        this.accountType = accountType;
        this.balance = balance;
    }
}

class SubSecureAccount extends SecureAccount {
    // 생성자가 부모와 같으면 생략 가능 (자동으로 부모 생성자 호출)
    constructor(owner: string, accountType: string, balance: number) {
        super(owner, accountType, balance);
    }
    getBalance(): number {
        return this.balance;        // 호출 여부와 상관없이 컴파일러가 이 줄을 읽고 에러 판단
    }
    getType(): string { 
        return this.accountType;    
    }
}

const myAcc = new SubSecureAccount("이영희", "정기적금", 30000);

// console.log(myAcc.accountType);   // 에러: protected 라서 외부 접근 불가
console.log(myAcc.getType());        // 에러: 자식 클래스의 getBalance() 단계에서 오류

/* 
타입스크립트는 코드 실행 전에 코드 전체를 정적으로 검사하기 때문에 본문 검사 기반으로 에러 반환
컴파일러는 이 메서드가 언제 불리는지를 따지지 않고 메서드 본문이 타입 규칙에 맞는지만 확인하기 때문
balance는 private이기 때문에 자식 클래스에서 접근할 수 없으니 호출이 되든 말든 호출 시도 자체가 에러
반면에 자바스크립트에서는 실행해야 에러가 발생 (호출하지 않으면 에러가 발생하지 않음)
*/
```


> [!NOTE] 필드와 메서드, 그리고 super와 this
> - 필드 ─ 값을 지정하는 변수 ─ `this.필드`형태로 접근 (접근 제한자가 허용하는 범위에서)
> - 메서드 ─ 동작하는 함수 ─ `super.메서드()` 형태로 접근 (오버라이딩 했을 때), `this.메서드()`도 가능
> 	- 자식 생성자에서는 `this`를 쓰기 전에 `super()`를 먼저 호출해야 한다. 
> 	- 메서드 안에서는 `super()`와 상관 없이 `this`를 쓸 수 있다. 
> - 생성자 호출 ─ `super(...)`

# 추상 클래스
- 함수의 기능 수행이 정의되지 않은 클래스 
	- 클래스 키워드(abstract)를 추가하면 선언할 수 있다. 
	- 미완성 클래스이기 때문에 `new` 로 만들 수 없다. 
	- 여러 자식 클래스가 같은 메서드 이름과 타입을 갖도록 보장할 수 있다. 
	- 공통 코드를 한 곳에 두어 중복을 피할 수 있다. 
- 추상 메서드 ─ 추상 클래스 안에서 이름과 타입만 정의하고 구현은 자식에게 강제로 맡긴다.
```ts
// 추상 클래스: 공통 틀만 정의하고, 직접 new 로 생성할 수 없습니다.
abstract class ServiceUser {

    // 일반 메서드: 자식이 그대로 물려받아 사용합니다.
    showProfile(): void {
        console.log("출력");
    }

    // 추상 메서드: 선언만 있고, 자식이 반드시 구현해야 합니다.
    abstract getRole(): string;
}

// 정상 자식 클래스
class CustomerUser extends ServiceUser {
    getRole(): string {
        return "일반 고객";
    }

    // 자식에만 있는 메서드 (오류 5에서 사용)
    customOnly(): string {
        return "고객 전용 기능";
    }
}

// 오류 1: 추상 클래스는 직접 생성할 수 없음
// const user1 = new ServiceUser();
// Cannot create an instance of an abstract class.

// 해결: 구현을 마친 자식 클래스로 생성
const user1 = new CustomerUser();

// 오류 2: 추상 메서드를 구현하지 않은 자식 클래스
// class BrokenUser extends ServiceUser { }
// Non-abstract class 'BrokenUser' does not implement inherited abstract member 'getRole'.

// 해결 A: 추상 메서드를 구현
class FixedUser extends ServiceUser {
    getRole(): string {
        return "구현 완료";
    }
}

// 해결 B: 자식도 추상 클래스로 만들면 구현을 미룰 수 있음 (단, 이 클래스도 new 불가)
abstract class StillAbstractUser extends ServiceUser { }

// 오류 3: 추상 메서드를 잘못된 타입으로 구현
// class WrongTypeUser extends ServiceUser {
//     getRole(): number { return 1; }
// }
// Property 'getRole' in type 'WrongTypeUser' is not assignable to the same property in base type.

// 해결: 부모의 반환 타입(string)에 맞춤
class RightTypeUser extends ServiceUser {
    getRole(): string {
        return "1";
    }
}

// 오류 4: 추상 메서드에 본문을 작성
// abstract class BadAbstract {
//     abstract getName(): string { return "이름"; }
// }
// Method 'getName' cannot have an implementation because it is marked abstract.

// 해결 A: 추상 메서드는 선언만
abstract class GoodAbstract {
    abstract getName(): string;
}
// 해결 B: 본문이 필요하면 abstract 를 빼고 일반 메서드로
class NormalClass {
    getName(): string { return "이름"; }
}

// 오류 5: 부모 타입 변수로 자식 전용 멤버 접근
const customer = new CustomerUser();
const asParent: ServiceUser = customer;   // 부모 타입으로 받는 건 OK
asParent.getRole();                       // OK: 부모에 선언된 메서드
// asParent.customOnly();                 // 에러: 부모에 선언된 메서드가 아님
// Property 'customOnly' does not exist on type 'ServiceUser'.

// 해결 A: 타입 가드(instanceof)로 자식 타입으로 좁히기
if (asParent instanceof CustomerUser) {
    console.log(asParent.customOnly());   // OK: 이 블록에서는 CustomerUser
}
// 해결 B: 처음부터 자식 타입으로 선언
const asChild: CustomerUser = customer;
console.log(asChild.customOnly());
```
## 추상클래스와 인터페이스

|         | 추상 클래스          | 인터페이스               |
| ------- | --------------- | ------------------- |
| 구현 코드   | 일반 메서드에 포함 가능   | 불가 (구조만 정의)         |
| 런타임 존재  | 있음 (클래스)        | 없음                  |
| 상속      | `extends` (하나만) | `implements` (여러 개) |
| 코드 물려받기 | 일반 메서드는 물려받음    | 없음                  |
| 구현 강제   | 추상 메서드만         | 모든 멤버               |
| 런타임     | 호출 이후에도 존재      | 호출 뒤 사라짐            |
