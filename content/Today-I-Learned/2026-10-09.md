'use client';

  

import { useState, useRef, RefObject } from 'react';

import TodoItem from './todoitem';

  

export interface TodoType {

    id: number;

    title: string;

    checked?: boolean;

    ref?: RefObject<HTMLLIElement | null>; // useRef(null)은 초기에 null이라 | null 필요

}

  

export default function Todo(): React.JSX.Element {

    const [items, setItems] = useState<TodoType[]>([]);

    const [title, setTitle] = useState<string>('');

  

    const actionCount = useRef<number>(0); // 컴포넌트 내부에서 사용할 값 정의

    const inputRef = useRef<HTMLInputElement>(null);

    const chileRef = useRef<HTMLLIElement>(null);

  

    // 입력창 값이 바뀔 때마다 title 상태에 반영

    const titleHandler = (e: React.ChangeEvent<HTMLInputElement>) => {

        setTitle(e.target.value);

    };

  

    // 등록 버튼(submit) 시: 기본 새로고침 막고, 목록에 추가 후 입력창 비우기

    const formHandler = (e: React.SubmitEvent<HTMLFormElement>) => {

        // 기본 새로고침 방지

        e.preventDefault();

        const id = Date.now();

        // items.push({ title, checked: false });

        // 상태 배열이니까 push 대신 concat 사용 ↓↓↓

        setItems(items.concat({ id, title, checked: false }));

        // 입력값 비우기(초기화)

        setTitle('');

        // 훅(hook)으로 지역변수 역할을 할 값

        actionCount.current++;

        // console.log('actionCount', actionCount); // { current: n } 객체가 출력됨

        console.log('actionCount', actionCount.current);

        // 자식(li) 요소: 마지막으로 그려진 항목이 담긴다 (등록 직전 상태)

        console.log('chileRef', chileRef.current);

        // 양식 작성 후 비우기(초기화)

        const inputEl = inputRef.current;

        inputEl?.focus();

    };

  

    return (

        <>

            <form onSubmit={formHandler}>

                <input

                    type="text"

                    name="title"

                    className="border"

                    value={title}

                    onChange={titleHandler}

                    ref={inputRef}

                />

                <button

                    type="submit"

                    className="border ml-1 px-3 cursor-pointer transition hover:bg-gray-700 active:scale-95 active:bg-gray-500"

                >

                    등록

                </button>

            </form>

            <ul>

                {items.length === 0 ? (

                    <li>할 일을 등록하세요.</li>

                ) : (

                    // items.map(({ id, title }) => <li key={id}>{title}</li>)

                    // ↑ <li>를 직접 그리던 방식 → TodoItem 컴포넌트로 대체

                    // key는 목록을 만드는 부모(map)에서 지정해야 한다

                    // items.map(({ id, title, checked }) => (

                    //     <TodoItem key={id} id={id} title={title} checked={checked} />

                    // ))

                    // ↑ 속성을 하나씩 나열하던 방식 → 전개 연산자(...)로 한 번에 전달

                    items.map((item) => (

                        <TodoItem key={item.id} {...item} ref={chileRef} />

                    ))

                )}

            </ul>

        </>

    );

}
