# js 함수
```javascript
function 함수명 (매개변수, ..) { // 함수명은 함수 객체의 이름표로서, 변수와 같다.
	(수행할 코드);
	return 반환값;
}
```
- 함수의 실행 평가 과정 ─ 실행 가능한 함수 객체를 만드는 과정 (실행 가능한 객체로 함수를 번역하는 과정)
	- 자바스크립트에서 순수 객체는 호출이 불가하고 별도 타입이나 인터페이스를 통해서 실행 가능하기 때문
	- *cf.* 실행 가능한 객체 (Execution Context)
	- 유효범위 체인 ─ 변수를 찾기 위해 안쪽 스코프에서 바깥쪽(상위) 스코프로 차례대로 찾아 나가는 연결 구조
		- 스코프(Scope) ─ 변수와 함수에 접근할 수 있는 유효한 범위
			- 함수가 중첩되면 스코프들도 계층적으로 중첩되며 외부 함수에서는 내부 함수의 변수에 접근할 수 없다. 
			- 내부 함수가 살아 있다면 스코프는 함수가 정의될 때 상위 스코프 참조를 (EC가 사라져도) 기억한다. 
		- 체인(Chain) ─ 스코프가 계층적으로 연결되어 리스트 형태를 이루는 것
		- 전역 스코프 도달 ─ 계속해서 찾아도 변수가 없으면 최종적으로 전역 스코프까지 확인
			- 스코프 체인이 끝(전역 스코프)까지 찾으려는 변수를 못 찾으면 `ReferenceError` 발생
	- this 바인딩 ─ 일반 함수를 호출한 객체 주소 ⇢ 정의할 때가 아닌 호출하는 시점에 결정된다. (화살표 함수 X)
# js 객체 상속
- 프로토타입 체인 ─ 프로토타입 기반의 객체 간 상속 관계를 나타내는 것 
	- 가장 최종 목적지는 Object ⇢ js 객체는 항상 상속 상태인데 그 대상은 Object인 셈
```javascript
const aObj = {a: 1, aa: 2}
const bObj = {b: 3, bb: 4}
const cObj = {c: 5, cc: 6}
bObj.__proto__ = aObj; // 상속 발생 ⇢ bObj는 aObj와 Object를 상속 받는다.
aObj.__proto__ = cObj; // 상속 발생 ⇢ aObj는 cObj와 Object를 상속 받는다. 
bObj.a  // 1
bObj.aa // 2
bObj.cc // 6 ⇢ bObj가 상속하는 aObj가 cObj를 상속하므로 bObj도 CObj를 상속하는 셈
```
- 생성자 함수 ─ 객체를 생성하는 함수 (이름 규칙으로 첫 글자를 대문자로 한다.) ─ *e.g.* new 연산자
	- 생성자 함수가 객체를 만드는 절차
		1. 생성자 함수의 prototype 객체를 상속한다.
		2. 생성자 함수의 this를 생성된 객체의 주소로 변경하고 실행한다.
```javascript
function Person() {
    this.name = "김철수";
    this.age = 20;
}                          // 아직 this는 뭔지 알 수 없음
const p1 = new Person();   // p1에 Person() 함수를 생성
p1;                        // this는 Person {name: '김철수', age: 20}을 가리킨다.
```
# 데이터 프로퍼티
- value ─ 값
- writable ─ 변경 가능 여부
- enumerable ─ 열거 가능 여부
	- for … in … 
	- object.keys() ─ false이면 객체의 속성이 노출되지 않음
- configurable ─ 재정의 가능 여부. 설정 값들을 바꿀 수 있는지 여부 ⇢ false 인 경우 delete 연산자로 삭제 불가
```js
const person = {
    name: "김철수",
    age: 20,
    showInfo() {
        console.log(this.name, this.age);
    }
};

Object.defineProperty(person, 'age', {writable: false}); // age 값 변경 불가로 변경
person.age = 30; // 조용히 무시됨
person; // {name: '김철수', age: 20}

Object.defineProperty(person, "showInfo", {enumerable: false}); // showinfo 열거 불가로 변경
Object.keys(person); // ['name', 'age']
for (const key in person) {console.log(key);} // name, age

Object.defineProperty(person, "name", {configurable: false}); // name 프로퍼티 삭제 불가로 변경
delete person.name; // false
```
# 접근자 프로퍼티
- getter와 setter ─ 이것만으로는 통제 안됨. 내부 속성 통제 및 은닉은 다른 기법(클로저+캡슐화)로 해야 한다. 
	- get 함수 ─ 읽기
	- set 함수 ─ 쓰기
```js
/* 언더스코어 캡슐화 */
const person = {
    _name: "김철수",   // "이 프로퍼티는 건들지 마세요." 라는 컨벤션일뿐 접근 제어가 아님
    _age: 20,         // 다시 말해, 이 프로퍼티는 getter/setter를 거쳐 써달라고 알려주는 용도
    get name() {
        console.log("get name()");
        return this._name;
    },
    set name(name) {
        console.log("set name()");
        this._name = name;
    },
    get age() {
        return this._age;
    },
    set age(age) {
        this._age = age;
    }
}

person; // {_name: '김철수', _age: 20} ⇢ 값이 바로 보임. 클로저와 다른 점! ⇢ 디버깅에 활용한다.
person.name;            // "get name()" 출력 후 '김철수'
person.name = "이영희"; // "set name()" 출력

/* 클로저 캡슐화 */
const person = (function () {
    let name = "김철수";     // 함수 종료 시 정리되어야 할 가비지컬렉션 대상
    let age = 20;           // 함수 종료 시 정리되어야 할 가비지컬렉션 대상
    return {
        get name() {         // 그런데 클로저가 가비지컬렉션 대상을 참조한다.
            return name;     // person 변수는 반환 값인 get name, … 객체를 참조하는데
        },                   // get name, … 객체들은 가비지컬렉션 대상을 참조하기 때문에 
        set name(value) {
            name = value;
        },
        get age() {
            return age;
        },
        set age(value) {
            age = value;
        }           
    }
})();

person;              // {} ⇢ 캡슐화 되었지만
person.name;         // '김철수' ⇢ getter/setter를 거치면 직접 접근할 수 있다.
person.name = "이영희";
person.name;         // '이영희' ⇢ 영희도 마찬가지! 프라이빗 변수를 흉내내는 셈 (물리제어)
```
- enumerabel ─ 열거 가능 여부
- configurable ─ 재정의 가능 여부. 설정 값들을 바꿀 수 있는지 여부

|             | **언더스코어**          | **클로저**      | **`#private`** |
| ----------- | ------------------ | ------------ | -------------- |
| 강제성         | 약속일 뿐, 실제 접근 막지 못함 | 물리적으로 접근 불가  | 물리적으로 접근 불가    |
| 디버깅         | 값이 보임              | `{}` 로만 보임   | 특수 표기로 보임      |
| `class`와 궁합 | 좋음                 | 어색함(팩토리 함수용) | 좋음(ES2022+)    |

# 직렬화 (Serializtion) 
- 전송 가능한 형태로 변환 (문자열로 변환)
	- js 에서는 전송 가능한 형태로 JSON 문자열을 쓴다. 
	- 함수, undefined 객체는 직렬화가 불가하고 null 값으로 전달한다. 
- JSON 객체
	1. 자바스크립트 객체에서 JSON 문자열로 직렬화 ─ stringify()
	2. JSON 문자열을 자바스크립트 객체로 역직렬화 ─ parse()
```js
const person = {name: "김철수", age: 20,};

/* 객체 동결 */
Object.freeze(person);
Object.getOwnPropertyDescriptors(person);
// { writable: false, configurable: false }

/* 직렬화 */
const jsonString = JSON.stringify(person);
jsonString;        // '{"name":"김철수","age":20}'
typeof jsonString; // 'string'

/* 역직렬화 */ 
const personStr = '{"name":"김철수","age":20}';
const person2 = JSON.parse(personStr);
person2; // {name: '김철수', age: 20}
```

> [!NOTE] 복사의 종류
> - 얕은 복사 ─ 주소만 복사
> - 깊은 복사 ─ 새로운 객체에 내용을 복사 (주소가 변경됨)
